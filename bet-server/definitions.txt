NFL player stat structure

url identifier = how betslips will retreive the data (special case for what can be inputted to identifier) (all identifiers should also accept o or u like o2.5 or u2.5)

"stats" : {
    "passing": {
        "C/ATT": "33/44",
        "YDS": 346,
        "AVG": 7.9,
        "TD": 2,
        "INT": 2,
        "SACKS": "3-19",
        "QBR": 66,
        "RTG": 93.6
    },
    "rushing": {
        "CAR": 3,
        "YDS": 16,
        "AVG": 5.3,
        "TD": 0,
        "LONG": 11
    },
    "fumbles": {
        "FUM": 2,
        "LOST": 1,
        "REC": 1
    }
    "defensive": {
        "TOT": 2,
        "SOLO": 1,
        "SACKS": 0,
        "TFL": 0,
        "PD": 0,
        "QB HTS": 0,
        "TD": 0
    }
    "receiving": {
        "REC": 1,
        "YDS": -6,
        "AVG": -6,
        "TD": 0,
        "LONG": -6,
        "TGTS": 1
    },
    "kicking": {
        "FG": "1/2",
        "PCT": 50,
        "LONG": 33,
        "XP": "2/2",
        "PTS": 5
    }
}

PYDS = passing.YDS
PATT = passing.C/ATT (strip the / and use the second number)
PCMP = passing.C/ATT (strip the / and use the first number)
PINT = passing.INT
PLNG = passing.LONG
PTD = passing.TD
TDS = rushing.TD + receiving.TD + defensive.TD ( TDS object can also accept yes or no like TDS=yes)
RYDS = rushing.YDS
PRYDS = passing.YDS + rushing.YDS
RLNG = rushing.LONG
RYDS = receiving.YDS
RLNG = receiving.LONG
RREC = receiving.REC
RRYDS = rushing.YDS + receiving.YDS
RATT = rushing.CAR
DSAC = defensive.SACKS
KXP = kicking.XP (strip the / and use the first number)
KFG = kicking.FG (strip the / and use the first number)
KPTS = kicking.PTS

total = add both teams total scores. Can be used with o/u like o48.5 or u48.5 (for NFL, NBA, NHL, UEFA)
1HPTS = add both teams 1st half scores. Can be used with o/u like o24.5 or u24.5 (same for 2HPTS) (for NFL and NBA)
1QPTS = add both teams 1st quarter scores. Can be used with o/u like o12.5 or u12.5 (same for 2QPTS, 3QPTS, 4QPTS) (for NFL and NBA)


NBA player stat structure

"1Q": {
    "PTS": 11,
    "REB": 0,
    "AST": 2
},
"stats": {
    "MIN": 36,
    "PTS": 29,
    "FG": "10-24",
    "3PT": "5-12",
    "FT": "4-5",
    "REB": 4,
    "AST": 8,
    "TO": 1,
    "STL": 2,
    "BLK": 0,
    "OREB": 2,
    "DREB": 2,
    "PF": 3,
    "+/-": -1
}

PRA = stats.PTS + stats.REB + stats.AST
AST = stats.AST
PA = stats.PTS + stats.AST
PR = stats.PTS + stats.REB
PTS = stats.PTS
REB = stats.REB
1QAST = 1Q.AST
1QPTS = 1Q.PTS
RA = stats.REB + stats.AST
1QREB = 1Q.REB
3PM = stats.3PT (strip the - and use the first number)
STL = stats.STL
BLK = stats.BLK
firstBasket = whether the player scored the first basket of the game (yes or no)
    obtained from   
    "firstBasket": {
        "athleteId": "5104157",
        "athleteName": "Victor Wembanyama",
        "teamId": "24",
        "team": "24",
        "scoreValue": 2,
        "period": 1
    },
    in base data (data.firstBasket)
2DBL = whether the player recorded a 2 double-double (yes or no) (points+rebounds, points+assists, rebounds+assists > 10 (steals and blocks count as well))
3DBL = whether the player recorded a 3 double-double (yes or no) (points+rebounds+assists > 10 (steals and blocks count as well))

moneylineReg = moneyline bet on the regular time result (only first 4 quarters in linescore) (can accept home or away abbreviation or "draw") (for NBA, NHL, UEFA)
points1Q = points scored by a team in the 1st quarter (same for points2Q, points3Q, points4Q) (for NBA)
points1H = points scored by a team in the 1st half (same for points2H) (for NBA)

UGL = stats.G (can also accept yes or no like GOAL=yes)
CARD = stats.YC + stats.RC (can also accept yes or no like CARD=yes)
RC = stats.RC

bothScore = whether both teams scored in the match (yes or no)
totalCorner = total number of corner kicks in the match (can be used with o/u like o10.5 or u10.5)
totalCards = total number of yellow and red cards in the match (can be used with o/u like o5.5 or u5.5)
moneyline1H = moneyline bet on the result at halftime (can accept home or away abbreviation or "draw". Same for moneyline2H) (for UEFA)
